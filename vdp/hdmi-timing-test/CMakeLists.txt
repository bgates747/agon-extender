cmake_minimum_required(VERSION 3.16)
file(READ "${CMAKE_CURRENT_LIST_DIR}/version.txt" PROJECT_VER)
string(STRIP "${PROJECT_VER}" PROJECT_VER)
set(COMPONENTS main)
if(AGON_HDMI_MODE0_PROBE)
    set(probe mode0)
elseif(AGON_HDMI_PAIR_PROBE)
    set(probe pair)
endif()
if(probe)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.pair.defaults")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(hdmi_timing)
if(probe)
    idf_component_get_property(main_target main COMPONENT_LIB)
    string(TOUPPER "${probe}" probe_upper)
    target_compile_definitions("${main_target}" PRIVATE AGON_HDMI_${probe_upper}_PROBE=1)
    include("${CMAKE_CURRENT_LIST_DIR}/pair/driver.cmake")
endif()
