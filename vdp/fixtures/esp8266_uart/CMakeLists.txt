cmake_minimum_required(VERSION 3.16)
foreach(name NET008_BUILD_ID NET008_SOURCE_COMMIT)
  if(NOT DEFINED ${name} OR "${${name}}" STREQUAL "")
    message(FATAL_ERROR "Explicit ${name} required")
  endif()
endforeach()
if(NOT NET008_BUILD_ID MATCHES "^[A-Za-z0-9._-]+$" OR
   NOT NET008_SOURCE_COMMIT MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "Invalid identified source/build")
endif()
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(net008_modem_probe)
if(NOT IDF_TARGET STREQUAL "esp32p4" OR NOT CONFIG_ESP_CONSOLE_USB_SERIAL_JTAG)
  message(FATAL_ERROR "P4 USB Serial/JTAG console required")
endif()
